# API compatibility allowlist for `make apicheck` (proposal 034).
#
# `make apicheck` compares every module's exported API against v0.2.2
# with apidiff and fails on any incompatible change not listed here.
#
# Proposal 034 rule: exported identifiers present in v0.2.2 are stable in
# signature and semantics. Do not remove, rename, or retype them, and do
# not add methods to exported interfaces. The only accepted incompatible
# change is appending a variadic `...Option` parameter to an existing
# function, and only when a call with no options behaves exactly as
# before (apart from defect fixes listed in the same proposal). Every
# entry here must also appear in engineering-docs/migration-v0.2-to-v0.30.md.
#
# Format: one entry per line, "<module path>: <apidiff line>", copied from
# the apicheck output, e.g.
#   github.com/jahkeup/corecbor/eat: DecodeClaims: changed from func([]byte) (*Claims, error) to func([]byte, ...Option) (*Claims, error)

# Proposal 027 H6: VerifyMulti fails closed; optional policy (RequireAny default).
github.com/jahkeup/corecbor/cose: VerifyMulti: changed from func(*Sign, []*Verifier) ([]int, error) to func(*Sign, []*Verifier, ...VerifyPolicy) ([]int, error)

# Proposal 027 H7: external AAD and algorithm options on multi-recipient
# verify/decrypt. No-option calls behave as v0.2.2 except header-selected
# HMAC 256/64, which is rejected (safety defect: attacker-chosen truncation).
github.com/jahkeup/corecbor/cose: DecryptMulti: changed from func(*Encrypt, KeyDeriver, int) ([]byte, error) to func(*Encrypt, KeyDeriver, int, ...EncryptOption) ([]byte, error)
github.com/jahkeup/corecbor/cose: VerifyMACMulti: changed from func(*Mac, KeyDeriver, int) error to func(*Mac, KeyDeriver, int, ...MACOption) error

# Proposal 033: NewInitiator/NewResponder gain ...Option
# (WithLegacyPreRFCProtocol). With no options they run RFC 9528, which is
# the contract-defect fix 033 makes; there is no other no-option change.
github.com/jahkeup/corecbor/edhoc: NewInitiator: changed from func(InitiatorConfig) (*Initiator, error) to func(InitiatorConfig, ...Option) (*Initiator, error)
github.com/jahkeup/corecbor/edhoc: NewResponder: changed from func(ResponderConfig) (*Responder, error) to func(ResponderConfig, ...Option) (*Responder, error)

# Proposal 032: eat entry points gain ...Option (WithLegacyDraftClaimKeys,
# CompatV02). With no options they use the IANA/RFC 9711 claim keys, which
# is the contract-defect fix 032 makes; the other no-option changes are
# the fail-closed submodule safety fix and unknown-claim preservation.
github.com/jahkeup/corecbor/eat: (*Claims).Encode: changed from func() ([]byte, error) to func(...Option) ([]byte, error)
github.com/jahkeup/corecbor/eat: DecodeClaims: changed from func([]byte) (*Claims, error) to func([]byte, ...Option) (*Claims, error)
github.com/jahkeup/corecbor/eat: Sign: changed from func(*Claims, *github.com/jahkeup/corecbor/cose.Signer) ([]byte, error) to func(*Claims, *github.com/jahkeup/corecbor/cose.Signer, ...Option) ([]byte, error)
github.com/jahkeup/corecbor/eat: Verify: changed from func([]byte, *github.com/jahkeup/corecbor/cose.Verifier) (*Claims, error) to func([]byte, *github.com/jahkeup/corecbor/cose.Verifier, ...Option) (*Claims, error)
